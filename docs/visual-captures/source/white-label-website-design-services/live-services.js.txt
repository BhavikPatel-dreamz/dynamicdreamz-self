  $(document).ready(function () {
    const $reviews = $('.review_animation');
    let currentIndex = 0;

    const starDelay = 200;
    const ratingDelay = 5 * starDelay + 300;
    const totalRatingDelay = 5 * starDelay + 600;

    // Function to reset all reviews animations and classes
    function resetAllReviews() {
      $reviews.each(function () {
        const $r = $(this);
        $r.removeClass('zoom-in show active');
        $r.find('.review_animation_wrapper').removeClass('show');
        $r.find('.review_animation_stars path').removeClass('show');
        $r.find('.review_ratings, .review_total_ratings').removeClass('show');
      });
    }

    function animateReview($review) {
      if (!$review.length) return;

      // No need to reset here anymore because we do it globally before calling animateReview

      setTimeout(() => {
        $review.addClass('zoom-in show');

        setTimeout(() => {
          $review.find('.review_animation_wrapper').addClass('show');

          $review.find('.review_animation_stars path').each((index, el) => {
            setTimeout(() => {
              $(el).addClass('show');
            }, index * starDelay);
          });

          setTimeout(() => {
            $review.find('.review_ratings').addClass('show');
          }, ratingDelay);

          setTimeout(() => {
            $review.find('.review_total_ratings').addClass('show');
          }, totalRatingDelay);

        }, 300);
      }, 100);
    }

    function showNextReview() {
      resetAllReviews();

      const $current = $reviews.eq(currentIndex);
      $current.addClass('active');
      animateReview($current);

      currentIndex = (currentIndex + 1) % $reviews.length;
    }

    showNextReview(); // Show first review on page load
    setInterval(showNextReview, 5000); // Rotate every 5 seconds
  });

jQuery(document).ready(function ($) {

  var counter_animated = false;

  if ('IntersectionObserver' in window) {

    var counter_observer = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (entry.isIntersecting && !counter_animated) {
          counter_animated = true;

          setTimeout(function () {
            $('#counter_1').html('20');
            $('#counter_2').html('150');
            $('#counter_3').html('2500');
            $('#counter_4').html('1000');
          }, 100);
        }
      });
    }, { threshold: 0.1 });

    var section = document.querySelector('.stats_sections');
    if (section) {
      counter_observer.observe(section);
    }

  } else {

    var scroll_timeout;

    $(window).on('scroll', function () {

      if (counter_animated) return;

      clearTimeout(scroll_timeout);

      scroll_timeout = setTimeout(function () {

        var section = $('.stats_section');
        if (!section.length) return;

        var window_top = $(window).scrollTop();
        var window_bottom = window_top + $(window).height();

        var section_top = section.offset().top;
        var section_bottom = section_top + section.outerHeight();

        if (window_bottom > section_top && window_top < section_bottom) {
          counter_animated = true;

          setTimeout(function () {
            $('#counter_1').html('18');
            $('#counter_2').html('150');
            $('#counter_3').html('2500');
            $('#counter_4').html('1000');
          }, 100);
        }

      }, 100);

    });

  }

});

function fixSlickAria($carousel) {
  $carousel.on('init reInit afterChange', function () {
    $(this).find('.slick-slide').each(function () {
      var isHidden = $(this).attr('aria-hidden') === 'true';

      $(this)
        .find('a, button, input, textarea, select, [tabindex]')
        .each(function () {
          if (isHidden) {
            $(this).attr('tabindex', '-1');
          } else {
            $(this).removeAttr('tabindex');
          }
        });
    });
  });
}

jQuery(document).ready(function ($) {

  $(".white_label_wp_development_service_accordion_section").each(function () {

    var $section = $(this);
    var $items = $section.find(".accordion_item");

    // Check current section only
    var isWebsiteDesign = $section.hasClass("website-design");

    // Default first accordion open for website-design
    if (isWebsiteDesign && !$items.filter(".active").length) {
      $items.first().addClass("active");
    }

    // Open active accordion on load
    $items.filter(".active").find(".accordion_body").each(function () {
      $(this).css("max-height", this.scrollHeight + "px");
    });

    // Initial icon setup
    $items.each(function () {

      if (!isWebsiteDesign) {

        // Normal accordion icons
        $(this)
          .find(".accordion_toggle")
          .text($(this).hasClass("active") ? "-" : "+");

      } else {

        // Remove text to avoid ++
        $(this).find(".accordion_toggle").text("");

      }

    });

    // Accordion click
    $section.find(".accordion_header").on("click", function () {

      var $current = $(this).closest(".accordion_item");
      var isActive = $current.hasClass("active");

      // Close all others
      $items.not($current)
        .removeClass("active")
        .find(".accordion_body")
        .css("max-height", 0);

      // Reset icons for normal accordions only
      if (!isWebsiteDesign) {

        $items.not($current)
          .find(".accordion_toggle")
          .text("+");

      }

      // Close current
      if (isActive) {

        $current
          .removeClass("active")
          .find(".accordion_body")
          .css("max-height", 0);

        if (!isWebsiteDesign) {
          $current.find(".accordion_toggle").text("+");
        }

      } else {

        // Open current
        $current.addClass("active");

        var body = $current.find(".accordion_body")[0];

        $(body).css("max-height", body.scrollHeight + "px");

        if (!isWebsiteDesign) {
          $current.find(".accordion_toggle").text("-");
        }
      }

    });

  });

jQuery(document).ready(function($) {
  $('.page-id-6865 .content-wrap a.btn.btn-red-brder').on('click', function(e) {
    var href = $(this).attr('href');

    if (href && href.charAt(0) === '#') {
      e.preventDefault();
      var $target = $(href);

      if ($target.length) {
        $('html, body').animate({

        }, 600); // duration in ms
      }
    }
  });
